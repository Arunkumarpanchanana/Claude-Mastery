MCP Sampling — LLMs Calling LLMs

MCP sampling is one of the most powerful — and least understood — MCP capabilities. It allows an MCP server to request an LLM completion from the client (Claude Code). In other words: your MCP tool can ask Claude questions.

This creates a powerful pattern: your MCP server handles specialized logic (database queries, API calls, computations), and when it needs to process language or reason about the results, it delegates back to Claude via sampling.

The sampling flow: Claude → calls MCP tool → MCP server processes → MCP server sends sampling/createMessage → Claude Code routes to LLM → LLM response returned to MCP server → MCP server returns tool result to Claude.

Why Sampling Matters

Without sampling, MCP servers are stateless data pipes — they take inputs and return outputs. With sampling, MCP servers become intelligent. They can:

  • Summarize large data retrieved from a database before returning it
  • Classify or categorize query results automatically
  • Generate structured output from unstructured raw data
  • Make decisions based on retrieved context

Implementing Sampling in Python

from mcp.server import Server
from mcp.server.models import InitializationOptions
import mcp.types as types

app = Server("intelligent-db-server")

@app.call_tool()
async def query_and_summarize(name: str, arguments: dict) -> list[types.TextContent]:
    """
    Tool that queries a database and uses sampling to summarize results.
    """
    if name != "query_and_summarize":
        raise ValueError(f"Unknown tool: {name}")

    sql_query = arguments.get("query", "")
    focus = arguments.get("focus", "key findings")

    # Step 1: Execute the database query (your actual DB logic here)
    raw_results = await execute_query(sql_query)  # Returns list of dicts

    if not raw_results:
        return [types.TextContent(type="text", text="No results found.")]

    # Step 2: Use sampling to have Claude summarize the results
    # This calls back to the MCP client (Claude Code) to get an LLM response
    sampling_result = await app.request_context.session.create_message(
        messages=[
            types.SamplingMessage(
                role="user",
                content=types.TextContent(
                    type="text",
                    text=f"""Summarize these database results focusing on: {focus}

Results:
{format_results(raw_results)}

Provide a concise summary with key insights and any anomalies."""
                )
            )
        ],
        max_tokens=500,
        system_prompt="You are a data analyst. Summarize database results concisely.",
        # The client (Claude Code) will use its default model for sampling
        # You can optionally suggest a model preference:
        model_preferences=types.ModelPreferences(
            hints=[types.ModelHint(name="claude-haiku-4-5")],  # Suggest fast model
            cost_priority=0.8  # Prefer cheaper models (0=quality, 1=cost)
        )
    )

    summary = sampling_result.content.text if hasattr(sampling_result.content, 'text') else str(sampling_result.content)

    return [types.TextContent(
        type="text",
        text=f"Query results ({len(raw_results)} rows):\n\n{summary}"
    )]

def format_results(results: list[dict]) -> str:
    if not results:
        return "No data"
    headers = list(results[0].keys())
    rows = [",".join(headers)]
    for r in results:
        rows.append(",".join(str(r[h]) for h in headers))
    return "\n".join(rows)

MCP Progress Notifications

For long-running operations, progress notifications let your MCP tool update the client on its status in real time. The client (Claude Code) can display these to the user.

from mcp.server import Server
import mcp.types as types

app = Server("progress-demo-server")

@app.call_tool()
async def process_large_dataset(name: str, arguments: dict) -> list[types.TextContent]:
    """Tool that processes data in stages and reports progress."""
    if name != "process_large_dataset":
        raise ValueError(f"Unknown tool: {name}")

    files = arguments.get("files", [])
    progress_token = arguments.get("_meta", {}).get("progressToken")

    total = len(files)
    results = []

    for i, file_path in enumerate(files):
        # Send progress notification
        if progress_token:
            await app.request_context.session.send_progress_notification(
                progress_token=progress_token,
                progress=i,
                total=total,
                # Optional message for display
            )

        # Process the file (your actual logic here)
        result = await process_file(file_path)
        results.append(result)

    # Send final progress (100%)
    if progress_token:
        await app.request_context.session.send_progress_notification(
            progress_token=progress_token,
            progress=total,
            total=total
        )

    return [types.TextContent(
        type="text",
        text=f"Processed {total} files. Results: {results}"
    )]

async def process_file(path: str) -> str:
    import asyncio
    await asyncio.sleep(0.1)  # Simulate work
    return f"processed:{path}"

MCP Log Notifications

Log notifications let your MCP server send structured log messages to the client. These appear in Claude Code's output and help with debugging.

import mcp.types as types

async def my_tool_handler(name: str, arguments: dict):
    """Send different levels of log messages."""
    session = app.request_context.session

    # Debug — detailed diagnostic info
    await session.send_log_message(
        level="debug",
        data={"action": "starting", "input": arguments},
        logger="my-tool"
    )

    try:
        result = await do_work(arguments)

        # Info — normal operation
        await session.send_log_message(
            level="info",
            data=f"Successfully processed {len(result)} items",
            logger="my-tool"
        )

        return result

    except Exception as e:
        # Error — something went wrong
        await session.send_log_message(
            level="error",
            data={"error": str(e), "traceback": "..."},
            logger="my-tool"
        )
        raise

Combining Sampling + Notifications

The most powerful pattern combines both: process data in stages (with progress notifications), use sampling to summarize intermediate results, and log key decisions:

@app.call_tool()
async def intelligent_pipeline(name: str, arguments: dict):
    """Full pipeline with progress, logging, and sampling."""
    session = app.request_context.session
    progress_token = arguments.get("_meta", {}).get("progressToken")
    stages = ["fetch", "preprocess", "analyze", "summarize"]

    for i, stage in enumerate(stages):
        # Progress notification
        if progress_token:
            await session.send_progress_notification(
                progress_token=progress_token,
                progress=i, total=len(stages)
            )

        # Log notification
        await session.send_log_message(
            level="info", data=f"Starting stage: {stage}", logger="pipeline"
        )

        if stage == "analyze":
            # Use sampling for intelligent analysis
            raw_data = await fetch_data()
            analysis = await session.create_message(
                messages=[types.SamplingMessage(
                    role="user",
                    content=types.TextContent(type="text", text=f"Analyze: {raw_data}")
                )],
                max_tokens=200
            )
            await session.send_log_message(
                level="debug", data=f"Analysis complete: {analysis.content}", logger="pipeline"
            )

    return [types.TextContent(type="text", text="Pipeline complete")]

Lab: Add Sampling and Progress Notifications to an MCP Server

Build a complete MCP server with a document analysis tool that uses sampling to summarize documents and sends progress notifications for multi-document processing.

#!/usr/bin/env python3
"""
mcp_advanced_server.py — MCP server with sampling + notifications.
Run with: python mcp_advanced_server.py
"""
import asyncio
import json
from mcp.server import Server
from mcp.server.stdio import stdio_server
import mcp.types as types

app = Server("document-analyzer")

@app.list_tools()
async def list_tools() -> list[types.Tool]:
    return [
        types.Tool(
            name="analyze_documents",
            description="Analyze multiple documents and produce a summary using AI.",
            inputSchema={
                "type": "object",
                "properties": {
                    "documents": {
                        "type": "array",
                        "items": {"type": "string"},
                        "description": "List of document texts to analyze"
                    },
                    "analysis_type": {
                        "type": "string",
                        "enum": ["summary", "sentiment", "themes"],
                        "default": "summary",
                        "description": "Type of analysis to perform"
                    }
                },
                "required": ["documents"]
            }
        )
    ]

@app.call_tool()
async def handle_tool(name: str, arguments: dict) -> list[types.TextContent]:
    if name != "analyze_documents":
        raise ValueError(f"Unknown tool: {name}")

    documents = arguments.get("documents", [])
    analysis_type = arguments.get("analysis_type", "summary")
    progress_token = arguments.get("_meta", {}).get("progressToken")
    session = app.request_context.session

    if not documents:
        return [types.TextContent(type="text", text="No documents provided.")]

    total = len(documents)
    individual_analyses = []

    await session.send_log_message(
        level="info",
        data=f"Starting {analysis_type} analysis of {total} documents",
        logger="document-analyzer"
    )

    # Process each document
    for i, doc in enumerate(documents):
        if progress_token:
            await session.send_progress_notification(
                progress_token=progress_token,
                progress=i,
                total=total + 1  # +1 for final synthesis step
            )

        await session.send_log_message(
            level="debug",
            data=f"Processing document {i+1}/{total} ({len(doc)} chars)",
            logger="document-analyzer"
        )

        # Use sampling to analyze each document
        analysis_prompt = {
            "summary": f"Summarize this document in 2-3 sentences:\n\n{doc}",
            "sentiment": f"Analyze the sentiment of this text (positive/negative/neutral) and explain why:\n\n{doc}",
            "themes": f"List the 3 main themes in this text as bullet points:\n\n{doc}"
        }.get(analysis_type, f"Analyze:\n\n{doc}")

        # Sampling: ask Claude to analyze the document
        sampling_result = await session.create_message(
            messages=[
                types.SamplingMessage(
                    role="user",
                    content=types.TextContent(type="text", text=analysis_prompt)
                )
            ],
            max_tokens=150,
            model_preferences=types.ModelPreferences(
                hints=[types.ModelHint(name="claude-haiku-4-5")],
                cost_priority=0.9  # Use cheap model for individual analyses
            )
        )

        analysis_text = sampling_result.content.text if hasattr(sampling_result.content, 'text') else str(sampling_result.content)
        individual_analyses.append({
            "document_index": i + 1,
            "analysis": analysis_text
        })

        await session.send_log_message(
            level="debug",
            data=f"Document {i+1} analysis: {analysis_text[:100]}...",
            logger="document-analyzer"
        )

    # Final synthesis step
    if progress_token:
        await session.send_progress_notification(
            progress_token=progress_token,
            progress=total,
            total=total + 1
        )

    await session.send_log_message(
        level="info",
        data="Synthesizing final report...",
        logger="document-analyzer"
    )

    # Use sampling again for synthesis
    analyses_text = "\n\n".join([
        f"Document {a['document_index']}: {a['analysis']}"
        for a in individual_analyses
    ])

    synthesis_result = await session.create_message(
        messages=[
            types.SamplingMessage(
                role="user",
                content=types.TextContent(
                    type="text",
                    text=f"""Synthesize these individual document analyses into a coherent overall report.
Identify common themes and key differences.

Individual analyses:
{analyses_text}

Write a unified analysis in 2-3 paragraphs."""
                )
            )
        ],
        max_tokens=400,
        model_preferences=types.ModelPreferences(
            hints=[types.ModelHint(name="claude-opus-4-5")],
            cost_priority=0.2  # Use quality model for synthesis
        )
    )

    synthesis = synthesis_result.content.text if hasattr(synthesis_result.content, 'text') else str(synthesis_result.content)

    # Final progress
    if progress_token:
        await session.send_progress_notification(
            progress_token=progress_token,
            progress=total + 1,
            total=total + 1
        )

    await session.send_log_message(
        level="info",
        data="Analysis complete!",
        logger="document-analyzer"
    )

    return [types.TextContent(
        type="text",
        text=f"Document Analysis Report ({analysis_type})\n{'='*40}\n\n{synthesis}\n\n"
             f"Individual analyses:\n" +
             "\n".join([f"- Doc {a['document_index']}: {a['analysis'][:100]}..."
                        for a in individual_analyses])
    )]

async def main():
    async with stdio_server() as (read_stream, write_stream):
        await app.run(
            read_stream,
            write_stream,
            InitializationOptions(
                server_name="document-analyzer",
                server_version="1.0.0",
                capabilities=app.get_capabilities(
                    notification_options=types.NotificationOptions(
                        tools_changed=False,
                        resources_changed=False,
                        prompts_changed=False
                    ),
                    experimental_capabilities={}
                )
            )
        )

if __name__ == "__main__":
    from mcp.server.models import InitializationOptions
    asyncio.run(main())
