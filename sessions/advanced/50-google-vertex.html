Claude on Google Vertex AI

Access Claude through Google Cloud's Vertex AI — learn project setup, the Vertex SDK, differences from the Anthropic SDK, and how to parallelize workloads.

45 min Advanced Code Lab 5 Quiz Questions

What is Google Vertex AI?

Google Vertex AI is GCP's managed machine learning platform. It includes access to foundation models including Claude (via the Model Garden), Gemini, and others. Like Bedrock for AWS, Vertex AI lets companies already in the GCP ecosystem access Claude through their existing Google Cloud infrastructure — IAM, VPC, billing, and compliance agreements.

Why Vertex AI for Claude? GCP-native data residency guarantees, Google Cloud's IAM and VPC integration, unified billing with other GCP services, integration with BigQuery and Cloud Storage for data pipelines, and Google's enterprise compliance certifications.

Project and Location Setup

Vertex AI organizes resources by GCP project and region (location). Claude is available in specific regions.

import anthropic

# The Anthropic SDK supports Vertex AI directly via the AnthropicVertex client
# Install: pip install anthropic[vertex]
# Or: pip install google-cloud-aiplatform anthropic

from anthropic import AnthropicVertex

# Setup
PROJECT_ID = "my-gcp-project"     # Your GCP project ID
LOCATION = "us-east5"              # Supported: us-east5, europe-west1, us-central1

client = AnthropicVertex(
    project_id=PROJECT_ID,
    region=LOCATION
)

Differences from the Anthropic SDK

Anthropic SDK

client = anthropic.Anthropic()
# Auth: ANTHROPIC_API_KEY env var

response = client.messages.create(
    model="claude-opus-4-5",
    max_tokens=1024,
    messages=[...]
)
  • Simple API key auth
  • Direct to Anthropic servers
  • model = "claude-opus-4-5"

Vertex AI SDK

client = AnthropicVertex(
    project_id="my-project",
    region="us-east5"
)
# Auth: Google ADC

response = client.messages.create(
    model="claude-opus-4-5@20251101",
    max_tokens=1024,
    messages=[...]
)
  • Google ADC / service account
  • Traffic within GCP network
  • model = "claude-...@version"

Vertex-Specific Features

  • Data residency: Choose regions where your data never leaves (EU, US, etc.) for GDPR compliance
  • VPC Service Controls: Restrict model access to specific VPC networks
  • Cloud Logging: All API calls automatically logged in Google Cloud Logging
  • Workload Identity: GKE workloads authenticate without managing credentials
  • Vertex AI Pipelines: Orchestrate Claude calls in Kubeflow-based ML pipelines
  • BigQuery integration: Process BigQuery datasets directly with Claude

Streaming on Vertex AI

from anthropic import AnthropicVertex

client = AnthropicVertex(project_id="my-project", region="us-east5")

# Streaming works identically to the Anthropic SDK
with client.messages.stream(
    model="claude-opus-4-5@20251101",
    max_tokens=500,
    messages=[{"role": "user", "content": "Write a short story about a robot learning to code."}]
) as stream:
    for text in stream.text_stream:
        print(text, end="", flush=True)

print(f"\n\nFinal message: {stream.get_final_message()}")

Parallelizing with Vertex AI

Vertex AI is designed for high-throughput, so parallelizing is a common pattern when processing large datasets stored in BigQuery or Cloud Storage:

import asyncio
from anthropic import AsyncAnthropicVertex

async def process_batch(items: list[str], project_id: str, region: str) -> list[str]:
    """Process multiple items in parallel on Vertex AI."""
    client = AsyncAnthropicVertex(project_id=project_id, region=region)

    async def process_one(item: str) -> str:
        response = await client.messages.create(
            model="claude-haiku-4-5@20241022",  # Use Haiku for speed/cost
            max_tokens=200,
            messages=[{"role": "user", "content": f"Summarize in one sentence: {item}"}]
        )
        return response.content[0].text

    # Fan-out: process all items concurrently
    tasks = [process_one(item) for item in items]
    results = await asyncio.gather(*tasks)
    return list(results)

# Example: process 10 items in parallel
async def main():
    items = [f"Document about topic {i}: Some content here..." for i in range(10)]
    results = await process_batch(items, "my-project", "us-east5")
    for i, r in enumerate(results, 1):
        print(f"{i}. {r}")

asyncio.run(main())

Lab: Make a Vertex AI API Call to Claude

Call Claude via Vertex AI and compare the response structure with a direct Anthropic SDK call. Also build a parallel document processor that uses Vertex AI.

"""
vertex_lab.py — Claude on Google Vertex AI
Requires: pip install anthropic[vertex]
Auth: gcloud auth application-default login
"""
import asyncio
import time
import os

# ── Configuration ──────────────────────────────────────────────────────────────
PROJECT_ID = os.getenv("GOOGLE_CLOUD_PROJECT", "your-project-id")
REGION = "us-east5"  # Primary Vertex region for Claude

def call_vertex_claude(prompt: str, model: str = "claude-haiku-4-5@20241022") -> dict:
    """Call Claude via Vertex AI."""
    try:
        from anthropic import AnthropicVertex
        client = AnthropicVertex(project_id=PROJECT_ID, region=REGION)

        start = time.time()
        response = client.messages.create(
            model=model,
            max_tokens=300,
            system="You are a concise assistant.",
            messages=[{"role": "user", "content": prompt}]
        )
        elapsed = time.time() - start

        return {
            "provider": "Vertex AI",
            "model": response.model,
            "text": response.content[0].text,
            "input_tokens": response.usage.input_tokens,
            "output_tokens": response.usage.output_tokens,
            "latency_ms": round(elapsed * 1000),
            "stop_reason": response.stop_reason
        }
    except ImportError:
        return {"provider": "Vertex AI", "error": "anthropic[vertex] not installed. Run: pip install anthropic[vertex]"}
    except Exception as e:
        return {"provider": "Vertex AI", "error": f"Error (check GCP auth + project): {str(e)[:150]}"}

def call_anthropic_direct(prompt: str) -> dict:
    """Call Claude via Anthropic SDK for comparison."""
    import anthropic
    client = anthropic.Anthropic()

    start = time.time()
    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=300,
        system="You are a concise assistant.",
        messages=[{"role": "user", "content": prompt}]
    )
    elapsed = time.time() - start

    return {
        "provider": "Anthropic Direct",
        "model": response.model,
        "text": response.content[0].text,
        "input_tokens": response.usage.input_tokens,
        "output_tokens": response.usage.output_tokens,
        "latency_ms": round(elapsed * 1000),
        "stop_reason": response.stop_reason
    }

# ── Comparison ─────────────────────────────────────────────────────────────────
def compare(prompt: str):
    print(f"\nPrompt: {prompt}")
    print("="*60)

    for fn in [call_anthropic_direct, call_vertex_claude]:
        result = fn(prompt)
        if "error" in result:
            print(f"\n[{result['provider']}] Error: {result['error']}")
        else:
            print(f"\n[{result['provider']}]")
            print(f"  Model: {result['model']}")
            print(f"  Latency: {result['latency_ms']}ms")
            print(f"  Tokens: {result['input_tokens']} in / {result['output_tokens']} out")
            print(f"  Response: {result['text'][:150]}")

compare("What is the main advantage of using Vertex AI for accessing Claude?")

# ── Parallel Processing Lab ────────────────────────────────────────────────────
async def parallel_vertex_processing():
    """Demonstrate parallel processing on Vertex AI."""
    try:
        from anthropic import AsyncAnthropicVertex
    except ImportError:
        print("\nSkipping parallel demo — anthropic[vertex] not installed")
        return

    ARTICLES = [
        "Quantum computing uses quantum mechanical phenomena to process information exponentially faster than classical computers.",
        "Neural networks are inspired by biological brains and consist of layers of interconnected nodes that learn from data.",
        "Blockchain technology provides decentralized, immutable ledgers that enable trustless transactions without intermediaries.",
        "CRISPR-Cas9 gene editing allows precise modification of DNA sequences, opening new possibilities in medicine and agriculture.",
    ]

    client = AsyncAnthropicVertex(project_id=PROJECT_ID, region=REGION)

    async def summarize(article: str, idx: int) -> tuple[int, str]:
        response = await client.messages.create(
            model="claude-haiku-4-5@20241022",
            max_tokens=100,
            messages=[{"role": "user", "content": f"One sentence summary: {article}"}]
        )
        return idx, response.content[0].text

    print("\n\nParallel document processing on Vertex AI:")
    start = time.time()

    tasks = [summarize(article, i) for i, article in enumerate(ARTICLES)]
    results = await asyncio.gather(*tasks)
    elapsed = time.time() - start

    results.sort(key=lambda x: x[0])
    for idx, summary in results:
        print(f"  Doc {idx+1}: {summary[:100]}")

    print(f"\nProcessed {len(ARTICLES)} docs in {elapsed:.1f}s")
    print(f"(Sequential would have been ~{len(ARTICLES)*2}s)")

asyncio.run(parallel_vertex_processing())
