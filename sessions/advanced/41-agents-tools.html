Agents with Tools & Environment Inspection

Build a fully autonomous agent with the complete observe-plan-act loop, environment inspection, in-context memory, and safe stopping conditions.

The Full Agentic Loop

An agent is a loop that runs until a stopping condition is met. Each iteration: Claude observes the current state, decides on an action (or decides to stop), executes the action via a tool, observes the result, and updates its plan.

Observe
Current state + history
Plan
Decide next action
Act
Call a tool
Done?
Check stopping conditions

Environment Inspection Strategy

Before taking action, a well-designed agent inspects its environment to understand the current state. This prevents wasted actions and reduces errors from incorrect assumptions.

import anthropic
import json
import os

client = anthropic.Anthropic()

# Define tools with rich descriptions that help Claude use them correctly
TOOLS = [
    {
        "name": "list_files",
        "description": "List files in a directory. Use this first to understand what's available before reading or writing.",
        "input_schema": {
            "type": "object",
            "properties": {
                "path": {"type": "string", "description": "Directory path to list. Use '.' for current directory."}
            },
            "required": ["path"]
        }
    },
    {
        "name": "read_file",
        "description": "Read the contents of a specific file.",
        "input_schema": {
            "type": "object",
            "properties": {
                "path": {"type": "string", "description": "File path to read"}
            },
            "required": ["path"]
        }
    },
    {
        "name": "write_file",
        "description": "Write content to a file. Creates the file if it doesn't exist.",
        "input_schema": {
            "type": "object",
            "properties": {
                "path": {"type": "string", "description": "File path to write"},
                "content": {"type": "string", "description": "Content to write"}
            },
            "required": ["path", "content"]
        }
    },
    {
        "name": "task_complete",
        "description": "Signal that the task is complete. Call this when you have finished the task.",
        "input_schema": {
            "type": "object",
            "properties": {
                "summary": {"type": "string", "description": "Brief summary of what was accomplished"}
            },
            "required": ["summary"]
        }
    }
]

def execute_tool(name: str, inputs: dict) -> str:
    """Execute a tool and return the result as a string."""
    if name == "list_files":
        try:
            files = os.listdir(inputs["path"])
            return json.dumps({"files": files, "count": len(files)})
        except Exception as e:
            return json.dumps({"error": str(e)})

    elif name == "read_file":
        try:
            with open(inputs["path"], "r") as f:
                return f.read()
        except Exception as e:
            return json.dumps({"error": str(e)})

    elif name == "write_file":
        try:
            with open(inputs["path"], "w") as f:
                f.write(inputs["content"])
            return json.dumps({"success": True, "path": inputs["path"]})
        except Exception as e:
            return json.dumps({"error": str(e)})

    elif name == "task_complete":
        return json.dumps({"done": True, "summary": inputs["summary"]})

The Agent Loop

def run_agent(task: str, max_iterations: int = 10) -> str:
    """
    Run the agentic loop until done or max iterations reached.

    Args:
        task: The task description for the agent
        max_iterations: Safety limit on number of steps

    Returns:
        Final summary from task_complete tool
    """
    messages = [{"role": "user", "content": task}]
    iterations = 0

    print(f"\nAgent starting: {task}")
    print("="*60)

    while iterations < max_iterations:
        iterations += 1
        print(f"\n--- Iteration {iterations} ---")

        # Claude observes state and plans next action
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1024,
            system="""You are an autonomous file agent. Follow this strategy:
1. ALWAYS start by listing files to understand what exists
2. Read relevant files before making decisions
3. Take one action at a time and observe results
4. When done, call task_complete with a summary""",
            tools=TOOLS,
            messages=messages
        )

        # Handle stop conditions
        if response.stop_reason == "end_turn":
            print("Agent: decided to stop (no tool use)")
            return response.content[0].text if response.content else "Done"

        if response.stop_reason != "tool_use":
            print(f"Agent: unexpected stop_reason: {response.stop_reason}")
            break

        # Process tool calls
        tool_results = []
        done = False

        for block in response.content:
            if block.type == "text":
                print(f"Claude thinks: {block.text}")
            elif block.type == "tool_use":
                print(f"Claude uses tool: {block.name}({json.dumps(block.input)[:80]}...)")
                result = execute_tool(block.name, block.input)
                print(f"Tool result: {result[:100]}...")

                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": result
                })

                if block.name == "task_complete":
                    done = True
                    summary = json.loads(result)["summary"]

        # Add assistant response and tool results to conversation
        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": tool_results})

        if done:
            print(f"\nAgent completed task: {summary}")
            return summary

    print(f"\nAgent hit max iterations ({max_iterations})")
    return "Max iterations reached"

# Run the agent
result = run_agent("Create a README.md file that summarizes what files exist in the current directory.")
print(f"\nFinal result: {result}")

Memory in Agents

Agents need memory to avoid repeating actions and to maintain context across steps. There are two main approaches:

In-Context Memory

  • The full conversation history IS the memory
  • Claude sees all previous actions and results
  • Simple — no external storage needed
  • Limited by context window size
  • Works for short-to-medium length tasks

External Memory

  • Store summaries or facts in a database
  • Retrieve relevant memories per step
  • Scales to arbitrarily long tasks
  • More complex to implement
  • Use for tasks spanning many sessions

Stopping Conditions

Without explicit stopping conditions, agents can loop indefinitely. Always implement:

  • Explicit done signal: A task_complete tool that the agent calls when finished
  • Max iterations: A hard limit on loop count (safety net)
  • Error threshold: Stop if the same tool fails 3+ times
  • Cost limit: Track token usage and stop when budget exceeded
  • Human checkpoint: Pause at key decision points for human approval

Lab: Build a Research Agent

Build an agent with three tools — search, summarize, and write — that researches a topic and produces a structured report.

import anthropic
import json

client = anthropic.Anthropic()

# Simulated search tool (in production, use a real search API)
KNOWLEDGE_BASE = {
    "claude api": "Claude API supports messages, tools, vision, streaming, and prompt caching. Models: Haiku (fast/cheap), Sonnet (balanced), Opus (most capable).",
    "rag": "RAG combines retrieval with generation. Documents are chunked, embedded, stored in vector DB, then retrieved at query time.",
    "prompt engineering": "Key techniques: role prompting, few-shot examples, chain-of-thought, XML tags for structure, explicit formatting instructions.",
    "function calling": "Tool use lets Claude call external functions. Define tools with JSON schema. Claude returns tool_use blocks, you execute and return tool_result.",
    "agent memory": "Agents use in-context memory (conversation history) or external memory (databases). In-context is simple but context-limited.",
}

RESEARCH_TOOLS = [
    {
        "name": "search",
        "description": "Search for information on a topic. Returns relevant information from the knowledge base.",
        "input_schema": {
            "type": "object",
            "properties": {
                "query": {"type": "string", "description": "Search query"}
            },
            "required": ["query"]
        }
    },
    {
        "name": "summarize",
        "description": "Summarize a long piece of text into key points.",
        "input_schema": {
            "type": "object",
            "properties": {
                "text": {"type": "string", "description": "Text to summarize"},
                "max_points": {"type": "integer", "description": "Maximum number of bullet points", "default": 3}
            },
            "required": ["text"]
        }
    },
    {
        "name": "write_report",
        "description": "Write the final research report. Call this when you have gathered enough information.",
        "input_schema": {
            "type": "object",
            "properties": {
                "title": {"type": "string"},
                "sections": {
                    "type": "array",
                    "items": {"type": "object", "properties": {
                        "heading": {"type": "string"},
                        "content": {"type": "string"}
                    }},
                    "description": "Report sections"
                }
            },
            "required": ["title", "sections"]
        }
    }
]

def execute_research_tool(name: str, inputs: dict) -> str:
    if name == "search":
        query = inputs["query"].lower()
        # Simple keyword matching against knowledge base
        results = []
        for key, value in KNOWLEDGE_BASE.items():
            if any(word in key for word in query.split()):
                results.append(f"[{key}]: {value}")
        return "\n\n".join(results) if results else "No results found for this query."

    elif name == "summarize":
        # In a real agent, this would call a separate API call
        # For demo, we call Claude inline
        r = client.messages.create(
            model="claude-haiku-4-5",
            max_tokens=200,
            messages=[{
                "role": "user",
                "content": f"Summarize in {inputs.get('max_points', 3)} bullet points:\n{inputs['text']}"
            }]
        )
        return r.content[0].text

    elif name == "write_report":
        report = f"# {inputs['title']}\n\n"
        for section in inputs['sections']:
            report += f"## {section['heading']}\n{section['content']}\n\n"
        return report

def run_research_agent(topic: str) -> str:
    messages = [{"role": "user", "content": f"Research this topic and write a structured report: {topic}"}]
    report = ""

    for i in range(8):  # Max 8 iterations
        response = client.messages.create(
            model="claude-opus-4-5",
            max_tokens=1500,
            system="""You are a research agent. Strategy:
1. Search for key aspects of the topic (2-3 searches)
2. Summarize what you've found
3. Write a structured report with 3-4 sections
Be methodical and thorough.""",
            tools=RESEARCH_TOOLS,
            messages=messages
        )

        if response.stop_reason == "end_turn":
            break

        tool_results = []
        for block in response.content:
            if block.type == "tool_use":
                print(f"[Tool: {block.name}] {str(block.input)[:80]}")
                result = execute_research_tool(block.name, block.input)
                if block.name == "write_report":
                    report = result
                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": result
                })

        messages.append({"role": "assistant", "content": response.content})
        messages.append({"role": "user", "content": tool_results})

        if response.stop_reason != "tool_use" or not any(b.type == "tool_use" for b in response.content):
            break

    return report or "Research complete (see conversation history)"

print(run_research_agent("How to build production-ready RAG systems with Claude"))
