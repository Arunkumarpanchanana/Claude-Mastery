Track 1: AI Research Assistant Architecture

# System components:

#    - Accept: PDF, URL, plain text
#    - Chunk with recursive splitter
#    - Add contextual retrieval (Claude Haiku)

#

#
# 3. Conversational layer
#    - Maintain chat history
#    - Allow follow-up questions

#    - Monitors progress via notifications
#    - Synthesizes results

# 2. Specialist subagents (claude-haiku-4-5)
#    - Data fetcher (MCP tool: fetch_data)
#    - Analyzer (MCP tool: run_analysis)
#    - Writer (MCP tool: write_report)
#    - Validator (MCP tool: check_quality)
#
# 3. Shared MCP server
#    - Provides tools available to all agents

Production Checklist

  • Rate limiting: Protect your API keys from runaway costs
  • Error handling: Graceful fallbacks when Claude API is unavailable
  • Prompt caching: Cache static system prompts for 60-80% cost reduction
  • Logging: Log all API calls with request IDs for debugging
  • Evals: At least 20 test cases measuring your core quality metric
  • User feedback: Thumbs up/down to collect signal for improvement
  • API key rotation: Store keys in environment variables, never source code
  • Cost monitoring: Set budget alerts in Anthropic console

Track 1 Capstone: AI Research Assistant (Complete Starter)

#!/usr/bin/env python3
"""
research_assistant.py — Complete AI Research Assistant
Capstone Track 1: RAG + Hybrid Search + Agent

Requirements:
    pip install anthropic sentence-transformers faiss-cpu rank-bm25 numpy
"""
import json
import re
import numpy as np
import faiss
import anthropic
from rank_bm25 import BM25Okapi
from sentence_transformers import SentenceTransformer, CrossEncoder
from dataclasses import dataclass, field

# ── Configuration ──────────────────────────────────────────────────────────────
EMBEDDER_MODEL = "all-MiniLM-L6-v2"
RERANKER_MODEL = "cross-encoder/ms-marco-MiniLM-L-6-v2"
CLAUDE_MODEL = "claude-opus-4-5"
HAIKU_MODEL = "claude-haiku-4-5"

client = anthropic.Anthropic()
embedder = SentenceTransformer(EMBEDDER_MODEL)
reranker = CrossEncoder(RERANKER_MODEL)

@dataclass
class Document:
    content: str
    source: str
    metadata: dict = field(default_factory=dict)

@dataclass
class Chunk:
    text: str
    source: str
    chunk_id: int
    contextualized_text: str = ""

class ResearchAssistant:
    """Full RAG-powered research assistant with hybrid search and reranking."""

    def __init__(self):
        self.chunks: list[Chunk] = []
        self.embeddings: np.ndarray | None = None
        self.faiss_index: faiss.IndexFlatIP | None = None
        self.bm25: BM25Okapi | None = None
        self.chat_history: list[dict] = []

    def chunk_document(self, text: str, chunk_size: int = 400, overlap: int = 60) -> list[str]:
        """Recursively chunk a document."""
        separators = ["\n\n", "\n", ". ", " "]

        def _split(text, seps):
            if not seps or len(text) <= chunk_size:
                return [text.strip()] if text.strip() else []
            sep = seps[0]
            parts = text.split(sep)
            chunks, current = [], ""
            for part in parts:
                candidate = current + (sep if current else "") + part
                if len(candidate) <= chunk_size:
                    current = candidate
                else:
                    if current.strip():
                        chunks.append(current.strip())
                    current = part if len(part) <= chunk_size else ""
                    if len(part) > chunk_size:
                        chunks.extend(_split(part, seps[1:]))
            if current.strip():
                chunks.append(current.strip())
            return chunks

        return _split(text, separators)

    def add_context_to_chunk(self, document: str, chunk: str, source: str) -> str:
        """Use Claude Haiku to add situating context."""
        try:
            response = client.messages.create(
                model=HAIKU_MODEL,
                max_tokens=80,
                system=[{
                    "type": "text",
                    "text": f"You contextualize document chunks. Source: {source}\n\nFull document:\n{document[:3000]}",
                    "cache_control": {"type": "ephemeral"}
                }],
                messages=[{"role": "user", "content": f"Write 1 sentence of context for this chunk:\n\n{chunk}"}]
            )
            return f"{response.content[0].text.strip()}\n\n{chunk}"
        except Exception:
            return chunk  # Fall back to original chunk on error

    def ingest(self, document: Document, use_contextual_retrieval: bool = True):
        """Ingest a document into the knowledge base."""
        print(f"Ingesting: {document.source}")
        raw_chunks = self.chunk_document(document.content)
        print(f"  Created {len(raw_chunks)} chunks")

        for i, chunk_text in enumerate(raw_chunks):
            if use_contextual_retrieval:
                contextualized = self.add_context_to_chunk(
                    document.content, chunk_text, document.source
                )
            else:
                contextualized = chunk_text

            self.chunks.append(Chunk(
                text=chunk_text,
                source=document.source,
                chunk_id=len(self.chunks),
                contextualized_text=contextualized
            ))

        self._rebuild_indexes()
        print(f"  Done. Total chunks in knowledge base: {len(self.chunks)}")

    def _rebuild_indexes(self):
        """Rebuild FAISS and BM25 indexes after ingestion."""
        texts = [c.contextualized_text for c in self.chunks]

        # Semantic index
        embeddings = embedder.encode(texts, normalize_embeddings=True).astype('float32')
        d = embeddings.shape[1]
        index = faiss.IndexFlatIP(d)
        index.add(embeddings)
        self.faiss_index = index
        self.embeddings = embeddings

        # BM25 index
        tokenized = [t.lower().split() for t in texts]
        self.bm25 = BM25Okapi(tokenized)

    def hybrid_search(self, query: str, top_k: int = 5) -> list[Chunk]:
        """Hybrid BM25 + semantic search with RRF and reranking."""
        if not self.chunks:
            return []

        # Semantic
        q_emb = embedder.encode([query], normalize_embeddings=True).astype('float32')
        _, sem_ids = self.faiss_index.search(q_emb, min(15, len(self.chunks)))
        sem_ids = [int(i) for i in sem_ids[0] if i >= 0]

        # BM25
        scores = self.bm25.get_scores(query.lower().split())
        bm25_ids = list(np.argsort(scores)[::-1][:15].astype(int))

        # RRF fusion
        rrf_scores = {}
        for rank, doc_id in enumerate(sem_ids):
            rrf_scores[doc_id] = rrf_scores.get(doc_id, 0) + 1.0 / (60 + rank + 1)
        for rank, doc_id in enumerate(bm25_ids):
            rrf_scores[doc_id] = rrf_scores.get(doc_id, 0) + 1.0 / (60 + rank + 1)

        candidates = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)[:15]
        candidate_chunks = [self.chunks[doc_id] for doc_id, _ in candidates]

        # Rerank
        if len(candidate_chunks) > top_k:
            pairs = [[query, c.contextualized_text] for c in candidate_chunks]
            rerank_scores = reranker.predict(pairs)
            ranked = sorted(zip(candidate_chunks, rerank_scores), key=lambda x: x[1], reverse=True)
            return [chunk for chunk, _ in ranked[:top_k]]

        return candidate_chunks[:top_k]

    def chat(self, user_message: str) -> str:
        """Conversational interface with RAG context."""
        # Retrieve relevant context
        retrieved = self.hybrid_search(user_message, top_k=4)

        if retrieved:
            context_parts = [f"[{c.source}]\n{c.text}" for c in retrieved]
            context = "\n\n---\n\n".join(context_parts)
            system = f"""You are a research assistant with access to a knowledge base.

Answer questions using the provided context. Always cite the source (shown in brackets).
If the context doesn't contain enough information, say so clearly and offer to search for more.

Context:
{context}"""
        else:
            system = "You are a helpful research assistant. The knowledge base is empty — ask the user to add documents."

        self.chat_history.append({"role": "user", "content": user_message})

        response = client.messages.create(
            model=CLAUDE_MODEL,
            max_tokens=800,
            system=system,
            messages=self.chat_history
        )

        assistant_message = response.content[0].text
        self.chat_history.append({"role": "assistant", "content": assistant_message})
        return assistant_message

# ── Demo ───────────────────────────────────────────────────────────────────────
def main():
    assistant = ResearchAssistant()

    # Ingest sample documents
    doc1 = Document(
        content="""
Transformer Architecture and Attention Mechanisms

The transformer architecture, introduced in 'Attention Is All You Need' (Vaswani et al., 2017),
revolutionized natural language processing. Its core innovation is the self-attention mechanism,
which allows each token to attend to all other tokens in the sequence simultaneously.

Self-attention computes three vectors for each token: Query (Q), Key (K), and Value (V).
The attention score between tokens is computed as: softmax(QK^T / sqrt(d_k)) * V
where d_k is the key dimension. This enables the model to learn which tokens are relevant
to each other regardless of their position in the sequence.

BERT uses bidirectional attention (sees all tokens) for understanding tasks.
GPT uses causal attention (only past tokens) for generation tasks.
Claude is based on a decoder-only transformer architecture similar to GPT.
""",
        source="transformer_paper_summary"
    )

    doc2 = Document(
        content="""
RAG Systems: Production Best Practices

Retrieval-Augmented Generation (RAG) combines retrieval systems with language models to
ground responses in factual documents. Key decisions in production RAG:

1. Chunking Strategy: Recursive chunking at natural boundaries outperforms fixed-size.
   Typical chunk size: 400-600 tokens with 10-15% overlap.

2. Embedding Models: sentence-transformers/all-MiniLM-L6-v2 for local/free use.
   Voyage AI or OpenAI embeddings for production quality.

3. Retrieval: Hybrid search (BM25 + semantic) outperforms either alone.
   BM25 wins on exact keyword matches; semantic wins on paraphrases.

4. Reranking: Cross-encoder reranking on top-20 candidates improves precision 30-50%.
   Use Cohere Rerank or sentence-transformers CrossEncoder.

5. Contextual Retrieval: Prepend situating context to each chunk before embedding.
   Anthropic reports 49% reduction in retrieval failures.
""",
        source="rag_best_practices"
    )

    assistant.ingest(doc1, use_contextual_retrieval=True)
    assistant.ingest(doc2, use_contextual_retrieval=True)

    # Interactive chat loop
    print("\n" + "="*60)
    print("AI Research Assistant Ready!")
    print("Type 'quit' to exit.")
    print("="*60)

    test_questions = [
        "How does self-attention work in transformers?",
        "What chunking strategy is recommended for production RAG?",
        "What is contextual retrieval and why does it help?"
    ]

    for q in test_questions:
        print(f"\nYou: {q}")
        answer = assistant.chat(q)
        print(f"Assistant: {answer}")
        print("-"*50)

if __name__ == "__main__":
    main()

Track 2 Quick Start: Code Review Bot

"""
Track 2 starter: GitHub webhook → multi-subagent review → PR comment

Architecture reference:
1. FastAPI webhook endpoint (POST /webhook/github)
2. Verify GitHub signature (X-Hub-Signature-256)
3. Extract PR diff from GitHub API
4. Run 3 subagents in parallel (security, performance, style)
5. Post synthesized review as PR comment via GitHub API

Key sessions to revisit:
- Session 42-43: Subagents (spawning, designing)
- Session 44: Claude Code Hooks (for CI integration)
- Session 40: Parallelization (running subagents concurrently)
"""
import anthropic
import hmac
import hashlib
import requests
from concurrent.futures import ThreadPoolExecutor, as_completed

GITHUB_TOKEN = "ghp_..."  # GitHub personal access token
WEBHOOK_SECRET = "..."     # Your webhook secret

client = anthropic.Anthropic()

def verify_signature(payload: bytes, signature: str) -> bool:
    expected = "sha256=" + hmac.new(
        WEBHOOK_SECRET.encode(), payload, hashlib.sha256
    ).hexdigest()
    return hmac.compare_digest(expected, signature)

def run_review_subagent(role: str, prompt: str, diff: str) -> dict:
    """Run one specialized reviewer."""
    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=600,
        system=f"You are a {role}. Review ONLY for your specialty. Be concise and specific.",
        messages=[{"role": "user", "content": f"{prompt}\n\nDiff:\n{diff}"}]
    )
    return {"role": role, "review": response.content[0].text}

def review_pr(diff: str) -> str:
    """Orchestrate multi-subagent PR review."""
    reviewers = [
        ("Security Specialist", "Find ONLY security vulnerabilities (injection, auth, crypto, secrets)."),
        ("Performance Analyst", "Find ONLY performance issues (N+1 queries, O(n²) algorithms, blocking I/O)."),
        ("Code Quality Reviewer", "Find ONLY maintainability issues (missing docs, unclear names, long functions)."),
    ]

    with ThreadPoolExecutor(max_workers=3) as executor:
        futures = [executor.submit(run_review_subagent, role, prompt, diff) for role, prompt in reviewers]
        results = [f.result() for f in as_completed(futures)]

    all_findings = "\n\n".join([f"=== {r['role']} ===\n{r['review']}" for r in results])

    synthesis = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=700,
        system="Synthesize specialist code reviews into a prioritized, actionable PR review.",
        messages=[{"role": "user", "content": f"Synthesize:\n{all_findings}"}]
    )
    return synthesis.content[0].text
