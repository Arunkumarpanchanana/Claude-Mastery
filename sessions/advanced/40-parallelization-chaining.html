Pattern Deep Dive: Parallelization

Parallelization runs multiple LLM calls simultaneously. Since each Claude API call typically takes 1-5 seconds, parallelizing 5 calls saves 4-20 seconds of wall-clock time. The tradeoff: more concurrent API calls means higher peak cost per second (though total cost stays the same).

Fan-out / Fan-in with asyncio

import asyncio
import anthropic
import time

client = anthropic.AsyncAnthropic()  # Use async client for concurrent calls

async def analyze_section(section: str, section_name: str) -> dict:
    """Analyze one document section asynchronously."""
    response = await client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=200,
        messages=[{
            "role": "user",
            "content": f"Summarize the key point of this section in 1-2 sentences:\n\n{section}"
        }]
    )
    return {"section": section_name, "summary": response.content[0].text}

async def parallel_document_analysis(sections: dict[str, str]) -> list[dict]:
    """Fan-out: analyze all sections in parallel. Fan-in: collect results."""
    tasks = [
        analyze_section(content, name)
        for name, content in sections.items()
    ]
    results = await asyncio.gather(*tasks)  # All run concurrently
    return list(results)

# Example usage
sections = {
    "Introduction": "RAG systems combine retrieval with generation...",
    "Methods": "We chunk documents using recursive splitting...",
    "Results": "Hybrid search improved precision by 34%...",
    "Conclusion": "Future work should explore reranking techniques..."
}

async def main():
    start = time.time()
    results = await parallel_document_analysis(sections)
    elapsed = time.time() - start
    print(f"Analyzed {len(results)} sections in {elapsed:.1f}s (parallel)")
    for r in results:
        print(f"\n[{r['section']}] {r['summary']}")

asyncio.run(main())
Speed comparison: 4 sequential calls at ~2s each = 8 seconds. 4 parallel calls = ~2 seconds. Parallelization provides near-linear speedup when tasks are I/O-bound (network calls), which API calls are.

Thread-based Parallelization (for synchronous contexts)

from concurrent.futures import ThreadPoolExecutor, as_completed
import anthropic

client = anthropic.Anthropic()  # Sync client

def analyze_section_sync(args):
    name, content = args
    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=200,
        messages=[{"role": "user", "content": f"Summarize: {content}"}]
    )
    return {"section": name, "summary": response.content[0].text}

def parallel_analysis(sections: dict, max_workers: int = 4) -> list[dict]:
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {
            executor.submit(analyze_section_sync, (name, content)): name
            for name, content in sections.items()
        }
        results = []
        for future in as_completed(futures):
            results.append(future.result())
    return results

Pattern Deep Dive: Prompt Chaining

Chaining passes the output of one LLM call as input to the next. Each step transforms the data in a focused way. Gates (conditional checks) can short-circuit the chain early.

Chaining with Gate Conditions

import anthropic
import json

client = anthropic.Anthropic()

def chain_with_gate(document: str) -> dict:
    """
    Chain: Extract → Validate → Translate
    Gate: If validation fails, stop early and return error.
    """
    # Step 1: Extract structured data
    step1 = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=300,
        messages=[{
            "role": "user",
            "content": f"""Extract from this text (respond ONLY with valid JSON):
{{
  "topic": "main topic in 5 words",
  "sentiment": "positive|negative|neutral",
  "key_claim": "most important claim",
  "has_data": true or false
}}

Text: {document}"""
        }]
    )

    try:
        extracted = json.loads(step1.content[0].text)
    except json.JSONDecodeError:
        return {"status": "error", "step": 1, "message": "Failed to parse JSON"}

    # GATE: Only proceed if document has data claims
    if not extracted.get("has_data"):
        return {"status": "skipped", "reason": "No quantitative data found", "data": extracted}

    # Step 2: Validate the key claim
    step2 = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=100,
        messages=[{
            "role": "user",
            "content": f"""Is this claim reasonable and verifiable?
Claim: {extracted['key_claim']}
Respond with only: VALID or INVALID, then one sentence why."""
        }]
    )
    validation = step2.content[0].text

    if "INVALID" in validation:
        return {"status": "rejected", "reason": validation, "data": extracted}

    # Step 3: Generate a headline
    step3 = client.messages.create(
        model="claude-opus-4-5",
        max_tokens=80,
        messages=[{
            "role": "user",
            "content": f"Write a professional news headline for this: {extracted['key_claim']}"
        }]
    )

    return {
        "status": "complete",
        "extracted": extracted,
        "validation": validation,
        "headline": step3.content[0].text.strip()
    }

result = chain_with_gate("Climate models predict 2.5°C warming by 2100 if current trends continue.")
print(json.dumps(result, indent=2))

Pattern Deep Dive: Routing

Routing classifies the input and dispatches it to a specialist handler. The classifier can be a fast/cheap model while handlers use an appropriate model per case.

import anthropic
import json

client = anthropic.Anthropic()

def classify_intent(message: str) -> str:
    """Fast classification using Haiku."""
    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=20,
        messages=[{
            "role": "user",
            "content": f"""Classify this message into exactly one category:
BILLING, TECHNICAL, ACCOUNT, GENERAL

Message: {message}
Category:"""
        }]
    )
    text = response.content[0].text.strip().upper()
    for cat in ["BILLING", "TECHNICAL", "ACCOUNT"]:
        if cat in text:
            return cat
    return "GENERAL"

# Specialist handlers (each has a focused system prompt)
HANDLERS = {
    "BILLING": {
        "system": "You are a billing specialist. Help with invoices, payments, and subscriptions. Be empathetic and solution-focused.",
        "model": "claude-haiku-4-5"
    },
    "TECHNICAL": {
        "system": "You are a senior technical support engineer. Diagnose and solve technical problems systematically. Ask clarifying questions if needed.",
        "model": "claude-opus-4-5"
    },
    "ACCOUNT": {
        "system": "You are an account management specialist. Handle account settings, permissions, and user management questions.",
        "model": "claude-haiku-4-5"
    },
    "GENERAL": {
        "system": "You are a helpful customer service representative. Answer general questions about the product.",
        "model": "claude-haiku-4-5"
    }
}

def route_and_handle(message: str) -> dict:
    """Classify and route to appropriate handler."""
    intent = classify_intent(message)
    handler = HANDLERS[intent]

    response = client.messages.create(
        model=handler["model"],
        max_tokens=300,
        system=handler["system"],
        messages=[{"role": "user", "content": message}]
    )

    return {
        "intent": intent,
        "model_used": handler["model"],
        "response": response.content[0].text
    }

# Test routing
test_messages = [
    "My invoice shows a charge I didn't authorize",
    "Getting a 503 error when I try to connect to your API",
    "How do I add a team member to my workspace?"
]

for msg in test_messages:
    result = route_and_handle(msg)
    print(f"\nMessage: {msg}")
    print(f"Routed to: {result['intent']} (using {result['model_used']})")
    print(f"Response: {result['response'][:100]}...")

Lab: Parallel Document Analyzer with Result Aggregator

Build a system that analyzes multiple documents in parallel, then aggregates and synthesizes the results into a single coherent report.

import asyncio
import anthropic
import time
from dataclasses import dataclass

client = anthropic.AsyncAnthropic()

DOCUMENTS = {
    "doc1": """
    Transformer architecture revolutionized NLP in 2017. The attention mechanism
    allows parallel processing of all tokens. BERT achieved state-of-the-art results
    on 11 NLP benchmarks. GPT-3's 175B parameters enabled few-shot learning.
    """,
    "doc2": """
    Large language models show emergent capabilities at scale. Chain-of-thought
    prompting enables complex reasoning. Constitutional AI improves alignment.
    RLHF training makes models more helpful and less harmful.
    """,
    "doc3": """
    Vector databases enable semantic search over millions of documents.
    RAG pipelines combine retrieval with generation for accuracy.
    Embedding models map text to dense vector representations.
    Approximate nearest neighbor search makes retrieval fast at scale.
    """,
    "doc4": """
    Agentic AI systems can complete multi-step tasks autonomously.
    Tool use lets models interact with external APIs and databases.
    Multi-agent systems distribute complex tasks across specialized models.
    Human-in-the-loop oversight is critical for high-stakes decisions.
    """
}

@dataclass
class DocumentAnalysis:
    doc_id: str
    topic: str
    key_insight: str
    complexity: str  # "introductory" | "intermediate" | "advanced"
    word_count: int

async def analyze_document(doc_id: str, content: str) -> DocumentAnalysis:
    """Analyze a single document asynchronously."""
    response = await client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=200,
        messages=[{
            "role": "user",
            "content": f"""Analyze this document and respond in this exact format:
TOPIC: (3-5 word topic)
INSIGHT: (single most important insight, one sentence)
COMPLEXITY: (introductory|intermediate|advanced)

Document:
{content}"""
        }]
    )
    text = response.content[0].text

    # Parse response
    def extract(label):
        for line in text.split('\n'):
            if line.startswith(label + ':'):
                return line.split(':', 1)[1].strip()
        return "unknown"

    return DocumentAnalysis(
        doc_id=doc_id,
        topic=extract("TOPIC"),
        key_insight=extract("INSIGHT"),
        complexity=extract("COMPLEXITY"),
        word_count=len(content.split())
    )

async def aggregate_analyses(analyses: list[DocumentAnalysis]) -> str:
    """Fan-in: synthesize all analyses into a coherent report."""
    analyses_text = "\n".join([
        f"Document {a.doc_id}: Topic='{a.topic}', Insight='{a.key_insight}', Level={a.complexity}"
        for a in analyses
    ])

    response = await client.messages.create(
        model="claude-opus-4-5",
        max_tokens=500,
        messages=[{
            "role": "user",
            "content": f"""You have analyzed {len(analyses)} documents about AI/ML.
Here are the individual analyses:

{analyses_text}

Write a 3-paragraph synthesis that:
1. Identifies the common themes across documents
2. Notes the progression from foundational to advanced topics
3. Recommends a learning order for someone new to the field"""
        }]
    )
    return response.content[0].text

async def main():
    print("Starting parallel document analysis...")
    start = time.time()

    # Fan-out: analyze all documents in parallel
    tasks = [analyze_document(doc_id, content)
             for doc_id, content in DOCUMENTS.items()]
    analyses = await asyncio.gather(*tasks)

    parallel_time = time.time() - start
    print(f"\nAll {len(analyses)} documents analyzed in {parallel_time:.1f}s")

    # Display individual results
    print("\n" + "="*50)
    print("INDIVIDUAL ANALYSES")
    print("="*50)
    for a in analyses:
        print(f"\n[{a.doc_id}] {a.topic}")
        print(f"  Insight: {a.key_insight}")
        print(f"  Complexity: {a.complexity}")

    # Fan-in: aggregate
    print("\n" + "="*50)
    print("SYNTHESIZED REPORT")
    print("="*50)
    report = await aggregate_analyses(analyses)
    print(report)

    total_time = time.time() - start
    print(f"\nTotal wall-clock time: {total_time:.1f}s")
    print(f"(Sequential would have been ~{len(DOCUMENTS) * 2}s)")

asyncio.run(main())
Pattern achieved: Fan-out (4 parallel analyze calls) + Fan-in (1 aggregation call using all results). Notice the separation: Haiku handles individual analysis (fast, cheap), Opus handles synthesis (complex reasoning). Right model for right job.
