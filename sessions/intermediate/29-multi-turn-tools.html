Multi-Turn Tool Conversations

Build the agentic loop — the core pattern for autonomous Claude agents that use tools over multiple turns to complete complex tasks.

The Agentic Loop Pattern

An agentic loop is a while loop that keeps calling Claude until stop_reason == "end_turn". Each iteration:

  1. Call Claude with current message history
  2. If stop_reason == "tool_use": execute all requested tools, append results, continue loop
  3. If stop_reason == "end_turn": Claude is done — return the final text
  4. Safety: track iteration count and break if max iterations exceeded
import anthropic

client = anthropic.Anthropic()

def run_agent(user_message: str, tools: list, max_iterations: int = 10) -> str:
    """Core agentic loop — runs until Claude stops or max iterations."""
    messages = [{"role": "user", "content": user_message}]

    for iteration in range(max_iterations):
        response = client.messages.create(
            model="claude-sonnet-4-5",
            max_tokens=4096,
            tools=tools,
            messages=messages
        )

        # Always append Claude's full response to history
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason == "end_turn":
            # Extract the text response
            for block in response.content:
                if hasattr(block, "text"):
                    return block.text
            return ""

        elif response.stop_reason == "tool_use":
            # Execute all tool calls in this response
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    result = execute_tool(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": str(result)
                    })
            messages.append({"role": "user", "content": tool_results})

    return "Max iterations reached without a final answer."

Multiple Tool Calls Per Turn

Claude can request multiple tools in a single response — always process all of them before the next API call:

# Claude's response might contain:
# [TextBlock("Let me check both cities..."), ToolUseBlock(name="get_weather", city="Paris"), ToolUseBlock(name="get_weather", city="Tokyo")]

tool_results = []
for block in response.content:
    if block.type == "tool_use":
        result = execute_tool(block.name, block.input)
        tool_results.append({
            "type": "tool_result",
            "tool_use_id": block.id,
            "content": str(result)
        })
# Send ALL results in one user message
messages.append({"role": "user", "content": tool_results})

Using Multiple Different Tools Together

The real power of agentic loops is chaining different tools. Claude might:

  1. Search for information (search tool)
  2. Do math on the results (calculator tool)
  3. Write the answer to a file (filesystem tool)

stop_reason: tool_use vs end_turn

stop_reasonMeaningAction
tool_useClaude wants to call one or more toolsExecute tools, send results, loop again
end_turnClaude has completed its responseExtract text and return to user
max_tokensResponse was cut offHandle as error or increase max_tokens
Safety: Always implement a maximum iteration limit. Bugs in tool implementations can cause Claude to loop indefinitely. 10-20 iterations is a reasonable cap for most agentic tasks.

Lab: Agent Using 3 Tools in an Agentic Loop

Build an agent with search, calculator, and note-taking tools that works through multi-step problems.

"""
agentic_loop_lab.py — Agent with 3 tools running an agentic loop
"""
import anthropic
import math
import json

client = anthropic.Anthropic()

# ── Simulated knowledge base ───────────────────────────────────────────────────
KNOWLEDGE_BASE = {
    "python speed": "Python runs at roughly 10-100 million operations per second on modern hardware.",
    "earth radius": "Earth's mean radius is 6,371 kilometers (3,959 miles).",
    "speed of light": "The speed of light in vacuum is 299,792,458 meters per second.",
    "population usa": "The US population is approximately 335 million people (2024 estimate).",
    "mount everest": "Mount Everest is 8,848.86 meters (29,031.7 feet) above sea level.",
}

NOTES = {}  # Simulated persistent notes store

# ── Tool definitions ───────────────────────────────────────────────────────────
TOOLS = [
    {
        "name": "search_knowledge",
        "description": "Search for factual information on a topic. Returns relevant data if found.",
        "input_schema": {
            "type": "object",
            "properties": {
                "query": {"type": "string", "description": "Search query keyword(s)"}
            },
            "required": ["query"]
        }
    },
    {
        "name": "calculate",
        "description": "Evaluate a mathematical expression. Use Python math syntax. Import math is available.",
        "input_schema": {
            "type": "object",
            "properties": {
                "expression": {"type": "string", "description": "Expression to evaluate, e.g. '6371 * 2 * math.pi'"}
            },
            "required": ["expression"]
        }
    },
    {
        "name": "save_note",
        "description": "Save a key finding or intermediate result for later reference.",
        "input_schema": {
            "type": "object",
            "properties": {
                "key": {"type": "string", "description": "Short identifier for the note"},
                "value": {"type": "string", "description": "The content to save"}
            },
            "required": ["key", "value"]
        }
    }
]


# ── Tool implementations ───────────────────────────────────────────────────────
def search_knowledge(query: str) -> str:
    query_lower = query.lower()
    for k, v in KNOWLEDGE_BASE.items():
        if any(word in query_lower for word in k.split()):
            return v
    return f"No data found for: {query}"

def calculate(expression: str) -> dict:
    try:
        result = eval(expression, {"__builtins__": {}}, {"math": math})
        return {"expression": expression, "result": round(result, 4)}
    except Exception as e:
        return {"error": str(e)}

def save_note(key: str, value: str) -> str:
    NOTES[key] = value
    return f"Saved note '{key}'"

def execute_tool(name: str, args: dict) -> str:
    if name == "search_knowledge":
        return search_knowledge(**args)
    elif name == "calculate":
        return str(calculate(**args))
    elif name == "save_note":
        return save_note(**args)
    return f"Unknown tool: {name}"


# ── Agentic loop ───────────────────────────────────────────────────────────────
def run_agent(task: str, verbose: bool = True) -> str:
    messages = [{"role": "user", "content": task}]
    iteration = 0
    max_iter = 10

    while iteration < max_iter:
        iteration += 1
        if verbose:
            print(f"\n[Iteration {iteration}] Calling Claude...")

        response = client.messages.create(
            model="claude-sonnet-4-5",
            max_tokens=2048,
            tools=TOOLS,
            messages=messages
        )

        messages.append({"role": "assistant", "content": response.content})

        if verbose:
            print(f"  stop_reason: {response.stop_reason}")

        if response.stop_reason == "end_turn":
            for block in response.content:
                if hasattr(block, "text"):
                    return block.text
            return ""

        elif response.stop_reason == "tool_use":
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    if verbose:
                        print(f"  Tool: {block.name}({json.dumps(block.input)[:60]})")
                    result = execute_tool(block.name, block.input)
                    if verbose:
                        print(f"  Result: {str(result)[:80]}")
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            messages.append({"role": "user", "content": tool_results})

    return "Max iterations reached."


# ── Test with multi-step tasks ─────────────────────────────────────────────────
tasks = [
    "What is the circumference of the Earth in kilometers? Show your calculation.",
    "How long would it take light to travel around the Earth? Use c = 299,792,458 m/s and Earth's circumference.",
]

for task in tasks:
    print(f"\n{'='*60}")
    print(f"TASK: {task}")
    print('='*60)
    answer = run_agent(task)
    print(f"\nFINAL ANSWER:\n{answer}")
    if NOTES:
        print(f"\nNotes saved: {NOTES}")
