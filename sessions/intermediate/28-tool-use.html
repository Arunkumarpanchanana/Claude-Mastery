Tool Use & Function Calling

Give Claude real capabilities — define tools it can call, execute them yourself, and feed results back. This is how Claude integrates with the real world.

What Is Tool Use?

Tool use (also called function calling) lets Claude request actions that it cannot perform itself — like looking up live data, querying a database, or running code. The flow is:

  1. You define tools with JSON Schema descriptions
  2. Claude decides if it needs a tool and responds with a tool_use block
  3. Your code executes the tool with the provided arguments
  4. You send the result back as a tool_result message
  5. Claude uses the result to formulate its final answer

Defining Tool Schemas

Tools are defined using JSON Schema. Be descriptive — Claude reads your descriptions to decide when and how to use the tool.

tools = [
    {
        "name": "get_weather",
        "description": "Get the current weather for a city. Returns temperature (Celsius), conditions, and humidity.",
        "input_schema": {
            "type": "object",
            "properties": {
                "city": {
                    "type": "string",
                    "description": "The city name, e.g. 'San Francisco' or 'Tokyo'"
                },
                "units": {
                    "type": "string",
                    "enum": ["celsius", "fahrenheit"],
                    "description": "Temperature units. Default: celsius"
                }
            },
            "required": ["city"]
        }
    }
]

Sending Tools to the API

import anthropic

client = anthropic.Anthropic()

response = client.messages.create(
    model="claude-sonnet-4-5",
    max_tokens=1024,
    tools=tools,
    messages=[{
        "role": "user",
        "content": "What's the weather like in Paris right now?"
    }]
)

print(response.stop_reason)  # "tool_use" — Claude wants to call a tool

Handling the tool_use Response

# Claude responded with a tool_use block
if response.stop_reason == "tool_use":
    tool_block = next(b for b in response.content if b.type == "tool_use")

    print(f"Tool name : {tool_block.name}")        # "get_weather"
    print(f"Tool ID   : {tool_block.id}")          # "toolu_01..."
    print(f"Arguments : {tool_block.input}")       # {"city": "Paris", "units": "celsius"}

Executing the Tool and Returning Results

def get_weather(city: str, units: str = "celsius") -> dict:

        "temperature": 18,
        "units": units,

# Execute the tool
tool_result = get_weather(**tool_block.input)

# Send the result back
final_response = client.messages.create(
    model="claude-sonnet-4-5",
    max_tokens=1024,
    tools=tools,
    messages=[
        # Original user message
        {"role": "user", "content": "What's the weather like in Paris right now?"},
        # Claude's tool call (must include ALL content blocks from the response)
        {"role": "assistant", "content": response.content},
        # Your tool result
        {
            "role": "user",
            "content": [
                {
                    "type": "tool_result",
                    "tool_use_id": tool_block.id,
                    "content": str(tool_result)
                }
            ]
        }
    ]
)

print(final_response.content[0].text)
# "The weather in Paris is currently 18°C and partly cloudy with 72% humidity."
Key insight: You execute the tool — not Claude. Claude only decides to call it and provides the arguments. This keeps you in control of what actions are actually taken.

Lab: Build a Weather Tool and Calculator Tool

Implement two tools and a complete one-turn tool-use loop.

Open in Colab 
"""
tool_use_lab.py — Weather + Calculator tools with full loop
"""
import anthropic
import math

client = anthropic.Anthropic()

# ── Tool definitions ──────────────────────────────────────────────────────────
TOOLS = [
    {
        "name": "get_weather",
        "description": "Get current weather for a city. Returns temperature, conditions, and wind speed.",
        "input_schema": {
            "type": "object",
            "properties": {
                "city": {"type": "string", "description": "City name"},
                "units": {
                    "type": "string",
                    "enum": ["celsius", "fahrenheit"],
                    "description": "Temperature units"
                }
            },
            "required": ["city"]
        }
    },
    {
        "name": "calculate",
        "description": "Evaluate a mathematical expression. Use Python math syntax.",
        "input_schema": {
            "type": "object",
            "properties": {
                "expression": {
                    "type": "string",
                    "description": "Math expression to evaluate, e.g. '2 ** 10', 'math.sqrt(144)'"
                }
            },
            "required": ["expression"]
        }
    }
]


# ── Tool implementations ──────────────────────────────────────────────────────
WEATHER_DATA = {
    "paris": {"temp_c": 18, "conditions": "Partly cloudy", "wind_kph": 15},
    "tokyo": {"temp_c": 28, "conditions": "Sunny", "wind_kph": 8},
    "london": {"temp_c": 12, "conditions": "Overcast", "wind_kph": 22},
    "new york": {"temp_c": 22, "conditions": "Clear", "wind_kph": 18},
}

def get_weather(city: str, units: str = "celsius") -> dict:
    data = WEATHER_DATA.get(city.lower())
    if not data:
        return {"error": f"No weather data for {city}"}
    temp = data["temp_c"] if units == "celsius" else round(data["temp_c"] * 9/5 + 32, 1)
    return {
        "city": city,
        "temperature": temp,
        "units": units,
        "conditions": data["conditions"],
        "wind_kph": data["wind_kph"]
    }

def calculate(expression: str) -> dict:
    try:
        result = eval(expression, {"__builtins__": {}}, {"math": math})
        return {"expression": expression, "result": result}
    except Exception as e:
        return {"error": str(e)}


def execute_tool(tool_name: str, tool_input: dict) -> str:
    """Route tool calls to the right implementation."""
    if tool_name == "get_weather":
        return str(get_weather(**tool_input))
    elif tool_name == "calculate":
        return str(calculate(**tool_input))
    return f"Unknown tool: {tool_name}"


def run_with_tools(user_message: str) -> str:
    """Run a single-turn conversation with tool use."""
    messages = [{"role": "user", "content": user_message}]

    response = client.messages.create(
        model="claude-sonnet-4-5",
        max_tokens=1024,
        tools=TOOLS,
        messages=messages
    )

    # Handle tool calls (may have multiple tools in one response)
    if response.stop_reason == "tool_use":
        # Append Claude's response (with tool_use blocks) to history
        messages.append({"role": "assistant", "content": response.content})

        # Execute all requested tools
        tool_results = []
        for block in response.content:
            if block.type == "tool_use":
                print(f"  [Tool call] {block.name}({block.input})")
                result = execute_tool(block.name, block.input)
                tool_results.append({
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": result
                })

        # Send results back
        messages.append({"role": "user", "content": tool_results})

        # Get final response
        final = client.messages.create(
            model="claude-sonnet-4-5",
            max_tokens=1024,
            tools=TOOLS,
            messages=messages
        )
        return final.content[0].text

    return response.content[0].text


# ── Test the tools ────────────────────────────────────────────────────────────
queries = [
    "What's the weather in Tokyo?",
    "What is the square root of 1764?",
    "If it's 28 degrees Celsius in Tokyo, what is that in Fahrenheit?",
]

for query in queries:
    print(f"\nUser: {query}")
    answer = run_with_tools(query)
    print(f"Claude: {answer}")
