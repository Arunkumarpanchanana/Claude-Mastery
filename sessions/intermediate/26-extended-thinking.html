Extended Thinking

Extended thinking gives Claude a scratchpad to reason through complex problems before producing its final answer. It's Claude's version of "showing its work" — except the thinking happens in a private block you can optionally inspect.

import anthropic

client = anthropic.Anthropic()

response = client.messages.create(
    model="claude-sonnet-4-5",  # Extended thinking requires a capable model
    max_tokens=16000,
    thinking={
        "type": "enabled",
        "budget_tokens": 10000  # How many tokens Claude can use for thinking
    },
    messages=[{
        "role": "user",
        "content": "A farmer has 17 sheep. All but 9 die. How many sheep are left? Explain your reasoning."
    }]
)

# Response has multiple content blocks
for block in response.content:
    if block.type == "thinking":
        print(f"[THINKING]\n{block.thinking}\n")
    elif block.type == "text":
        print(f"[ANSWER]\n{block.text}")

When to Use Extended Thinking

  • Mathematical or logical proofs that require many steps
  • Complex code architecture decisions
  • Analyzing trade-offs with multiple competing factors
  • Tasks where you've noticed reasoning errors in standard mode
Cost note: Thinking tokens count toward your token usage. Set budget_tokens conservatively (1000-4000 for most tasks; 10000+ only for genuinely hard problems). The thinking content itself is not billed at the same rate as output — check current pricing.

Image Input

Claude can process images sent as base64-encoded data or as public URLs:

import base64

# Option 1: URL (simpler, for publicly accessible images)
response = client.messages.create(
    model="claude-sonnet-4-5",
    max_tokens=512,
    messages=[{
        "role": "user",
        "content": [
            {
                "type": "image",
                "source": {
                    "type": "url",
                    "url": "https://upload.wikimedia.org/wikipedia/commons/thumb/4/47/PNG_transparency_demonstration_1.png/280px-PNG_transparency_demonstration_1.png"
                }
            },
            {"type": "text", "text": "Describe what you see in this image."}
        ]
    }]
)

# Option 2: Base64 (for local files or private images)
with open("chart.png", "rb") as f:
    image_data = base64.standard_b64encode(f.read()).decode("utf-8")

response = client.messages.create(
    model="claude-sonnet-4-5",
    max_tokens=1024,
    messages=[{
        "role": "user",
        "content": [
            {
                "type": "image",
                "source": {
                    "type": "base64",
                    "media_type": "image/png",  # image/jpeg, image/gif, image/webp
                    "data": image_data
                }
            },
            {"type": "text", "text": "What trend does this chart show?"}
        ]
    }]
)
print(response.content[0].text)

PDF Support via Files API

Claude can read PDFs natively. Upload a PDF once via the Files API and reference it in multiple requests:

# Step 1: Upload the PDF
with open("annual_report.pdf", "rb") as f:
    file_response = client.beta.files.upload(
        file=("annual_report.pdf", f, "application/pdf"),
    )
file_id = file_response.id

# Step 2: Reference the file in a message
response = client.beta.messages.create(
    model="claude-sonnet-4-5",
    max_tokens=2048,
    messages=[{
        "role": "user",
        "content": [
            {
                "type": "document",
                "source": {
                    "type": "file",
                    "file_id": file_id
                }
            },
            {"type": "text", "text": "What are the three main risks mentioned in this report?"}
        ]
    }],
    betas=["files-api-2025-04-14"]
)
print(response.content[0].text)

# Step 3: Delete the file when done
client.beta.files.delete(file_id)
Files API benefit: Uploading a PDF once and reusing the file_id for multiple questions saves you from resending the same large document repeatedly, reducing cost and latency.

Lab: Vision, PDFs, and Extended Thinking

Three standalone code snippets you can run independently to explore each capability.

"""
multimodal_lab.py — Extended Thinking, Vision, and PDF demos
"""
import anthropic
import base64
import httpx

client = anthropic.Anthropic()


# ── Lab 1: Extended Thinking ──────────────────────────────────────────────────
def demo_extended_thinking():
    print("\n=== Extended Thinking Demo ===")
    response = client.messages.create(
        model="claude-sonnet-4-5",
        max_tokens=8000,
        thinking={
            "type": "enabled",
            "budget_tokens": 3000
        },
        messages=[{
            "role": "user",
            "content": (
                "You have a 3-gallon jug and a 5-gallon jug. "
                "Measure exactly 4 gallons of water. Show your work."
            )
        }]
    )

    for block in response.content:
        if block.type == "thinking":
            # Show first 300 chars of thinking
            print(f"Thinking preview: {block.thinking[:300]}...")
        elif block.type == "text":
            print(f"Answer: {block.text}")


# ── Lab 2: Image Understanding ────────────────────────────────────────────────
def demo_image_url():
    print("\n=== Image URL Demo ===")
    # Use a well-known public image
    image_url = "https://upload.wikimedia.org/wikipedia/commons/thumb/6/6e/Python_logo_and_wordmark.svg/800px-Python_logo_and_wordmark.svg.png"

    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=256,
        messages=[{
            "role": "user",
            "content": [
                {
                    "type": "image",
                    "source": {"type": "url", "url": image_url}
                },
                {
                    "type": "text",
                    "text": "What programming language logo is this? Describe its colors and design."
                }
            ]
        }]
    )
    print(response.content[0].text)


def demo_image_base64():
    print("\n=== Image Base64 Demo ===")
    # Download an image and convert to base64
    image_data = base64.standard_b64encode(
        httpx.get("https://upload.wikimedia.org/wikipedia/commons/thumb/6/6e/Python_logo_and_wordmark.svg/800px-Python_logo_and_wordmark.svg.png").content
    ).decode("utf-8")

    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=256,
        messages=[{
            "role": "user",
            "content": [
                {
                    "type": "image",
                    "source": {
                        "type": "base64",
                        "media_type": "image/png",
                        "data": image_data
                    }
                },
                {"type": "text", "text": "In 2 sentences, describe this image."}
            ]
        }]
    )
    print(response.content[0].text)


# ── Lab 3: Multiple images in one message ────────────────────────────────────
def demo_image_comparison():
    print("\n=== Multi-Image Comparison Demo ===")
    urls = [
        "https://upload.wikimedia.org/wikipedia/commons/thumb/6/6e/Python_logo_and_wordmark.svg/800px-Python_logo_and_wordmark.svg.png",
        "https://upload.wikimedia.org/wikipedia/commons/thumb/1/18/ISO_C%2B%2B_Logo.svg/640px-ISO_C%2B%2B_Logo.svg.png",
    ]

    content = []
    for i, url in enumerate(urls, 1):
        content.append({"type": "image", "source": {"type": "url", "url": url}})
        content.append({"type": "text", "text": f"Image {i} above."})

    content.append({"type": "text", "text": "Compare these two programming language logos. What is different about their design philosophies?"})

    response = client.messages.create(
        model="claude-haiku-4-5",
        max_tokens=256,
        messages=[{"role": "user", "content": content}]
    )
    print(response.content[0].text)


if __name__ == "__main__":
    demo_extended_thinking()
    demo_image_url()
    demo_image_base64()
    demo_image_comparison()
