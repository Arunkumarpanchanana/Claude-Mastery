The MCP Python SDK

Anthropic provides an official Python SDK for building MCP servers. Install it with:

pip install mcp

Basic MCP Server Structure

from mcp.server.fastmcp import FastMCP

# Create the server with a name
mcp = FastMCP("My First MCP Server")

# Run with stdio transport (for Claude Desktop)
if __name__ == "__main__":
    mcp.run()

Defining Tools with @mcp.tool()

Use the @mcp.tool() decorator. The function's docstring becomes the tool description that Claude reads:

from mcp.server.fastmcp import FastMCP

mcp = FastMCP("Calculator Server")

@mcp.tool()
def add(a: float, b: float) -> float:
    """Add two numbers together. Returns the sum."""
    return a + b

@mcp.tool()
def get_word_count(text: str) -> dict:
    """Count the number of words and characters in a text string.

    Returns a dictionary with 'words' and 'characters' counts.
    """
    words = len(text.split())
    chars = len(text)
    return {"words": words, "characters": chars}

@mcp.tool()
def reverse_string(text: str) -> str:
    """Reverse a string. E.g. 'hello' becomes 'olleh'."""
    return text[::-1]

Defining Resources with @mcp.resource()

Resources are read-only data that Claude can access. They have a URI scheme:

@mcp.resource("config://app-settings")
def get_app_settings() -> str:
    """Application configuration settings."""
    return """
APP_NAME=MyApp
VERSION=1.2.0
MAX_CONNECTIONS=100
LOG_LEVEL=INFO
"""

@mcp.resource("data://team-members")
def get_team_members() -> str:
    """List of current team members and their roles."""
    members = [
        {"name": "Alice Chen", "role": "Engineering Lead"},
        {"name": "Bob Kumar", "role": "Product Manager"},
        {"name": "Carol Smith", "role": "Designer"},
    ]
    import json
    return json.dumps(members, indent=2)

Defining Prompts with @mcp.prompt()

Prompts are reusable message templates that users can invoke by name in MCP clients:

from mcp.types import Message

@mcp.prompt()
def code_review_prompt(code: str, language: str = "python") -> list[Message]:
    """Generate a code review prompt for the given code snippet."""
    return [
        {
            "role": "user",
            "content": f"""Please review this {language} code:

```{language}
{code}
```

Check for:
1. Correctness and edge cases
2. Performance issues
3. Security vulnerabilities
4. Code style and readability
"""
        }
    ]

Lab: Build an MCP Server with 2 Tools

Build a complete, runnable MCP server with a text analysis tool and a URL info tool.

MCP Server Quickstart 
"""
my_mcp_server.py — A simple MCP server with 2 tools and 1 resource
pip install mcp
Run: python my_mcp_server.py
"""
from mcp.server.fastmcp import FastMCP
import json
import re
from datetime import datetime
from urllib.parse import urlparse

mcp = FastMCP("Text & URL Toolkit")


# ── Tool 1: Text Analysis ─────────────────────────────────────────────────────
@mcp.tool()
def analyze_text(text: str) -> dict:
    """Analyze a text string and return statistics.

    Returns word count, sentence count, average word length,
    most common words, and readability indicators.
    """
    words = text.split()
    sentences = re.split(r'[.!?]+', text)
    sentences = [s.strip() for s in sentences if s.strip()]

    word_freq: dict[str, int] = {}
    for word in words:
        clean = re.sub(r'[^a-zA-Z]', '', word).lower()
        if clean and len(clean) > 3:
            word_freq[clean] = word_freq.get(clean, 0) + 1

    top_words = sorted(word_freq.items(), key=lambda x: x[1], reverse=True)[:5]
    avg_word_len = sum(len(w) for w in words) / len(words) if words else 0

    return {
        "word_count": len(words),
        "sentence_count": len(sentences),
        "character_count": len(text),
        "average_word_length": round(avg_word_len, 1),
        "top_words": dict(top_words),
        "estimated_reading_time_seconds": round(len(words) / 238 * 60)
    }


# ── Tool 2: URL Info Extractor ─────────────────────────────────────────────────
@mcp.tool()
def extract_url_info(url: str) -> dict:
    """Extract and parse information from a URL.

    Returns the protocol, domain, path, query parameters, and fragment.
    Also indicates if the URL appears to be HTTPS-secured.
    """
    try:
        parsed = urlparse(url)
        query_params = {}
        if parsed.query:
            for param in parsed.query.split("&"):
                if "=" in param:
                    k, v = param.split("=", 1)
                    query_params[k] = v

        return {
            "original_url": url,
            "protocol": parsed.scheme,
            "domain": parsed.netloc,
            "path": parsed.path or "/",
            "query_parameters": query_params,
            "fragment": parsed.fragment or None,
            "is_secure": parsed.scheme == "https",
            "is_valid": bool(parsed.scheme and parsed.netloc)
        }
    except Exception as e:
        return {"error": str(e), "original_url": url}


# ── Resource: Server Status ─────────────────────────────────────────────────────
@mcp.resource("status://server-info")
def get_server_info() -> str:
    """Current server status and capabilities."""
    return json.dumps({
        "name": "Text & URL Toolkit",
        "version": "1.0.0",
        "started_at": datetime.now().isoformat(),
        "tools": ["analyze_text", "extract_url_info"],
        "resources": ["status://server-info"],
        "description": "Provides text analysis and URL parsing capabilities."
    }, indent=2)


# ── Test the tools locally (not via MCP) ────────────────────────────────────
def test_locally():
    """Quick test without MCP client."""
    print("=== analyze_text ===")
    result = analyze_text(
        "The quick brown fox jumps over the lazy dog. "
        "Python is a great programming language for building tools. "
        "MCP servers are easy to build with the FastMCP library."
    )
    print(json.dumps(result, indent=2))

    print("\n=== extract_url_info ===")
    for url in [
        "https://api.example.com/v1/users?page=2&limit=10#results",
        "http://localhost:8080/health",
        "not-a-url",
    ]:
        result = extract_url_info(url)
        print(f"\n{url}")
        print(json.dumps(result, indent=2))


if __name__ == "__main__":
    import sys
    if "--test" in sys.argv:
        test_locally()
    else:
        # Run as MCP server (stdio transport for Claude Desktop)
        mcp.run()
